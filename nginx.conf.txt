server {
    listen 8081;
    
    # 1. Menyajikan halaman antarmuka (UI)
    location / {
        root /usr/share/nginx/html;
        index index.html;
    }

    # 2. Meneruskan request API ke backend internal
    location /api/ {
        proxy_pass http://backend:5000/api/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}
